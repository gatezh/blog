{{- /* llms-full.txt — every page's text in one document, for agents that
       prefer a single fetch over crawling.

       Headings are shifted down two levels so each page nests under its own
       H2. Shortcodes are removed from .RawContent; a cfimage's caption (or
       alt) is kept as italic text, since that prose is real content, while
       the image link itself is dropped — this is a text corpus. */ -}}
{{- define "agentText" -}}
{{- $c := .RawContent -}}
{{- $c = replaceRE `(?s)\{\{<\s*cfimage\s+[^>]*?caption="([^"]*)"[^>]*>\}\}` "*$1*" $c -}}
{{- $c = replaceRE `(?s)\{\{<\s*cfimage\s+[^>]*?alt="([^"]*)"[^>]*>\}\}` "*$1*" $c -}}
{{- $c = replaceRE `(?s)\{\{[<%].*?[>%]\}\}` "" $c -}}
{{- $c | replaceRE `(?m)^(#{1,4}) ` "$1## " -}}
{{- end -}}

{{- end }}
