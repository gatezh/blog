{{- /* _headers — response headers for every page and file, applied by Workers
       static assets with no Worker code involved, so page views stay free and
       unmetered.
       https://developers.cloudflare.com/workers/static-assets/headers/

       The Content Security Policy is a host allowlist. Every script is a file
       served from this site (assets/js, built with js.Build and loaded with
       SRI) or comes from one of the third parties below, so script-src needs
       neither 'unsafe-inline' nor a per-response nonce:

       - Google Analytics (production builds only, like the tag in head.html):
         Google's policy for "Google Analytics without any Ads features" —
         script www.googletagmanager.com; img www.googletagmanager.com and
         *.google-analytics.com; connect those plus *.google.com — with
         *.googletagmanager.com in place of www. (a superset). The wildcards
         are required: collection goes to regional hosts such as
         region1.google-analytics.com, and when that hit fails gtag retries at
         www.google.com/g/collect.
         https://developers.google.com/tag-platform/security/guides/csp
       - Remark42 (params.remark42Host): its embed script renders the comment
         thread in an iframe from that host.
       - Cloudflare Turnstile (only when a site key is configured): api.js and
         the challenge iframe.
       - PostHog (only when params.posthogKey is set, i.e. production): its
         documented policy. The wildcard is required — PostHog warns that "the
         exact subdomains may change over time". script-src covers the array.js
         and recorder the snippet injects; session replay compresses in a Web
         Worker created from a blob, hence worker-src.
         https://posthog.com/docs/advanced/content-security-policy
         Violations of this policy are reported to the same PostHog project
         (report-uri for browsers without the Reporting API, report-to plus a
         Reporting-Endpoints header for those with it), so vendor drift that
         starts blocking a request shows up there. The host must match
         api_host in js/posthog.ts (US cloud). The token is public by design.
         https://posthog.com/docs/csp-tracking

       Each third-party origin is derived from the same param or condition the
       templates use to load it, so the policy cannot allow less than the page
       needs, and omits a service the build does not include. connect-src takes
       the contact API's origin from params.apiUrl, the URL the form posts to.

       style-src keeps 'unsafe-inline' for posts: Chroma writes code
       highlighting as inline style attributes (markup.highlight.noClasses
       defaults to true), and the cfimage shortcode emits a <style> block and
       style attributes. The theme and Turnstile need none of it.

       There is no upgrade-insecure-requests: every URL the site emits is
       relative or https, the zone redirects http to https, and under
       `wrangler dev` it would rewrite same-origin requests to an https port
       nothing listens on.

       Outside production (staging builds with HUGO_ENVIRONMENT=staging) every
       response also gets X-Robots-Tag: noindex: staging pages carry their own
       host as canonical, so without it staging is a crawlable duplicate of the
       whole site. It stays inside the one /* rule, because a second rule with
       the same pattern replaces the first rather than merging with it. Rules
       with different patterns do merge, which is what the agent-text rules at
       the end rely on.

       Same one-line-printf idiom as _redirects and llms.txt: this format is
       isPlainText, so the template IS the output, and gotmplfmt indents the
       body of a block with a literal tab. A header line has to be indented and
       a path line must not be. Limits: 100 rules, 2,000 characters per line.
       Wrangler skips a longer line with only a warning, which would silently
       drop the CSP, so the build fails instead. */ -}}
{{- $script := slice "'self'" -}}
{{- $img := slice "'self'" "data:" -}}
{{- $connect := slice "'self'" -}}
{{- $frame := slice -}}
{{- $worker := slice -}}
{{- $report := "" -}}
{{- with site.Params.apiUrl -}}
	{{- with urls.Parse . -}}
		{{- if .Host }}{{ $connect = $connect | append (printf "%s://%s" .Scheme .Host) }}{{ end -}}
	{{- end -}}
{{- end -}}
{{- if site.Params.comments -}}
	{{- with site.Params.remark42Host -}}
		{{- with urls.Parse . -}}
			{{- $remark42 := printf "%s://%s" .Scheme .Host -}}
			{{- $script = $script | append $remark42 -}}
			{{- $connect = $connect | append $remark42 -}}
			{{- $frame = $frame | append $remark42 -}}
		{{- end -}}
	{{- end -}}
{{- end -}}
{{- if site.Params.turnstileSiteKey -}}
	{{- $script = $script | append "https://challenges.cloudflare.com" -}}
	{{- $frame = $frame | append "https://challenges.cloudflare.com" -}}
{{- end -}}
{{- if site.Params.posthogKey -}}
	{{- $script = $script | append "https://*.posthog.com" -}}
	{{- $connect = $connect | append "https://*.posthog.com" -}}
	{{- $worker = slice "'self'" "blob:" "data:" -}}
	{{- $report = printf "https://us.i.posthog.com/report/?token=%s" (urlquery site.Params.posthogKey) -}}
{{- end -}}
{{- if and hugo.IsProduction site.Config.Services.GoogleAnalytics.ID -}}
	{{- $script = $script | append "https://*.googletagmanager.com" -}}
	{{- $img = $img | append (slice "https://*.google-analytics.com" "https://*.googletagmanager.com") -}}
	{{- $connect = $connect | append (slice "https://*.google-analytics.com" "https://*.google.com" "https://*.googletagmanager.com") -}}
{{- end -}}
{{- $csp := slice "default-src 'self'" (printf "script-src %s" (delimit $script " ")) "style-src 'self' 'unsafe-inline'" (printf "img-src %s" (delimit $img " ")) (printf "connect-src %s" (delimit $connect " ")) -}}
{{- with $frame }}{{ $csp = $csp | append (printf "frame-src %s" (delimit . " ")) }}{{ end -}}
{{- with $worker }}{{ $csp = $csp | append (printf "worker-src %s" (delimit . " ")) }}{{ end -}}
{{- $csp = $csp | append (slice "object-src 'none'" "base-uri 'none'" "form-action 'self'" "frame-ancestors 'none'") -}}
{{- with $report }}{{ $csp = $csp | append (slice (printf "report-uri %s" .) "report-to posthog") }}{{ end -}}
{{- $cspLine := printf "  Content-Security-Policy: %s" (delimit $csp "; ") -}}
{{- if gt (len $cspLine) 2000 }}{{ errorf "_headers: the Content-Security-Policy line is %d characters; Workers static assets skips any line over 2,000, which would drop the policy" (len $cspLine) }}{{ end -}}
{{- printf "/*\n" -}}
{{- printf "  X-Content-Type-Options: nosniff\n" -}}
{{- printf "  Referrer-Policy: strict-origin-when-cross-origin\n" -}}
{{- printf "  Permissions-Policy: camera=(), geolocation=(), microphone=(), payment=(), usb=()\n" -}}
{{- /* Superseded by frame-ancestors in current browsers; kept for older ones. */ -}}
{{- printf "  X-Frame-Options: DENY\n" -}}
{{- printf "%s\n" $cspLine -}}
{{- with $report }}{{ printf "  Reporting-Endpoints: posthog=%q\n" . }}{{ end -}}
{{- if not hugo.IsProduction }}{{ printf "  X-Robots-Tag: noindex\n" }}{{ end -}}
{{- /* Agent-facing text: the Markdown mirror of every page and the llms*.txt
       indexes. Crawlable, so an agent can fetch them, but never indexed —
       otherwise every page ships a full-text duplicate at a second URL.
       The splat rule matches mirrors at any depth but not the home page's,
       hence the separate /index.md rule. */ -}}
{{- range slice "/index.md" "/*/index.md" "/llms.txt" "/llms-full.txt" -}}
	{{- printf "\n%s\n  X-Robots-Tag: noindex\n" . -}}
{{- end -}}
{{- /* Fingerprinted assets: the content hash is in the file name, so a changed
       file is a new URL and the old one never changes — cache it for a year.
       js.html fingerprints every script in every build; css.html fingerprints
       the stylesheet only in production, so staging's /css/main.css keeps the
       default revalidation. /brand/ holds the favicons and share image, which
       head.html always fingerprints. /fonts/ and /favicon.ico are plain static/
       files and are left alone. */ -}}
{{- $immutable := slice "/js/*" "/brand/*" -}}
{{- if hugo.IsProduction }}{{ $immutable = $immutable | append "/css/*" }}{{ end -}}
{{- range $immutable -}}
	{{- printf "\n%s\n  Cache-Control: public, max-age=31536000, immutable\n" . -}}
{{- end -}}
